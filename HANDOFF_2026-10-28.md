### 2.3 Найден и доказан главный баг ссылок Getgems (НЕ исправлен в коде)

EQ   EQDHPnW_ADfklEsTL2Hyhvk4hzzoI8ihxfKIBhEuNY-Wxo6d

def friendly_address(raw: str) -> str:
    """raw '0:<hex>' -> user-friendly 'EQ...' (base64url + crc16-xmodem)."""
    wc_str, hex_part = raw.split(":", 1)
    payload = bytes([0x11]) + (b"\xff" if int(wc_str) == -1 else b"\x00") + bytes.fromhex(hex_part)
    crc = 0
    for b in payload:
        crc ^= b << 8
        for _ in range(8):
            crc = ((crc << 1) ^ 0x1021) & 0xFFFF if crc & 0x8000 else (crc << 1) & 0xFFFF
    return base64.urlsafe_b64encode(payload + crc.to_bytes(2, "big")).decode().rstrip("=")
